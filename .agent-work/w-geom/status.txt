## Ledger residual (on-device)

```sh
ROOT=/media/fat/misterplex_v2
LED=$ROOT/misterplexd.frame_ledger
echo "=== ledger path ==="; ls -la "$LED"
echo "=== last 30 events ==="; tail -30 "$LED"
echo "=== residual identity over ALL session_end rows ==="
awk '
  /event=session_end/ {
    f=0;p=0;d=0;r=0
    for(i=1;i<=NF;i++){
      split($i,a,"=")
      if(a[1]=="frames")f=a[2]+0
      if(a[1]=="presents")p=a[2]+0
      if(a[1]=="drops")d=a[2]+0
      if(a[1]=="residual")r=a[2]+0
    }
    sf+=f; sp+=p; sd+=d; sr+=r; n++
    if (r != f-p-d) bad++
  }
  /event=process_start/ { starts++ }
  /event=process_exit/  { exits++ }
  END {
    print "sessions=" n " process_starts=" starts+0 " process_exits=" exits+0
    print "sum_frames=" sf " sum_presents=" sp " sum_drops=" sd " sum_residual=" sr
    print "identity_check sum(frames-presents-drops)=" (sf-sp-sd) " vs sum(residual)=" sr
    print "row_bad_residual=" bad+0
    if (n==0) { print "EMPTY_LEDGER"; exit 2 }
    if (bad>0 || (sf-sp-sd)!=sr) { print "RESIDUAL_FAIL"; exit 1 }
    print "RESIDUAL_OK"
  }
' "$LED"; echo "true rc=$?"
echo "=== restarts visible ==="; grep -c event=process_start "$LED"; grep event=process_exit "$LED" | tail -5
```
