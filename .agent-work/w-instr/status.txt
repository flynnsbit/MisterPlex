## Parent capture commands (PTS-capable)

OUT=$HOME/misterplex-captures/pts_$(date +%Y%m%d_%H%M%S)
mkdir -p "$OUT"
# 1) START GRABBER FIRST (startup window), keep v4l2 timestamps
ffmpeg -hide_banner -f v4l2 -input_format mjpeg -video_size 1920x1080 \
  -i /dev/video0 -vsync 0 -copyts -c:v copy -t 120 \
  -f matroska "$OUT/cap.mkv"
# (cast while ffmpeg runs)
# 2) PTS sidecar (one float seconds per frame, index=line)
ffprobe -v error -select_streams v:0 \
  -show_entries frame=pkt_pts_time -of csv=p=0 \
  "$OUT/cap.mkv" > "$OUT/pts.csv"
# 3) PNG for OCR (same order)
ffmpeg -v error -i "$OUT/cap.mkv" -vsync 0 "$OUT/f_%04d.png"
# 4) wall measure
WALL=$(ffprobe -v error -show_entries format=duration -of csv=p=0 "$OUT/cap.mkv")
# 5) score with split
python3 tools/hdmi_motion_instrument.py "$OUT" \
  --source-fps 24/1 --capture-wall-s "$WALL" --pts-csv "$OUT/pts.csv"
