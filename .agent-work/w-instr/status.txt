### 1) Capture one tier (repeat for 240p and 480p conf)

# A) MKV+PTS preferred (enables device vs grabber split)
OUT=captures/glass_480_$(date +%Y%m%d_%H%M%S)
mkdir -p "$OUT"
# start grabber BEFORE cast; discard warm-up in tool (default 15 via OCR path)
/usr/bin/time -f 'wall_s=%e' -o "$OUT/wall.txt" \
  ffmpeg -y -v error \
    -use_wallclock_as_timestamps 1 \
    -f v4l2 -input_format mjpeg -video_size 1920x1080 -framerate 30 -i /dev/video0 \
    -an -c:v copy -t 90 -copyts -start_at_zero \
    "$OUT/cap.mkv"
# true rc of ffmpeg:
# ffmpeg ...; echo "ffmpeg true rc=$?"

ffprobe -v error -select_streams v:0 -show_entries frame=pkt_pts_time \
  -of csv=p=0 "$OUT/cap.mkv" > "$OUT/pts.csv"
ffmpeg -v error -i "$OUT/cap.mkv" -start_number 0 "$OUT/f_%04d.png"
