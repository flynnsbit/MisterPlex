  MISTERPLEX_FABRIC_DMA=1 ./fabric_content_match_lab --len 65536 --bank 1
  # optional:
  #   --force-lab   instead of env
  #   --bank-phys 0x...
  #   --cmd-seq N --polls N --start-w
  #   --ring-plxk   (default OFF; present after PLXH done only)

Exit codes:
  0  full content_match (PLXH Success + bank==staging)
  1  PLXH fail and/or content mismatch
  2  gate closed / bad args / product flag ON
  3  mmap /dev/mem or map failure
