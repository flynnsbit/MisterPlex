# Decode-core seam debt - measured, not asserted.
#
# `stream_path` instantiates `h264_decode_core` as `product_decode_core`, but
# instantiation is not connection.  As measured on 2026-07-28 (branch
# w-decode-o5), 30 of the core's inputs are tied to constants and ALL 12 of its
# outputs terminate in the synthesis `_keep` anti-prune wire.  The core is
# therefore product-*present* and product-*vacuous*: it cannot decode, and
# nothing it produces can be observed.  Every pixel written to the frame store
# still comes from `decode_stub`.
#
# `scripts/check_decode_core_seam.py` measures this from the RTL and requires an
# EXACT match with this file.  New vacuity fails the gate; removing vacuity also
# fails the gate until the corresponding line is deleted here, so progress is
# always recorded and never silently absorbed.
#
# DO NOT add entries to make a gate pass.  Delete entries by wiring real signals.

[presentation_driver]
# The module that actually drives stream_path fs_wr_en/fs_wr_pixel/fs_wr_reset/
# fs_swap.  This must become h264_decode_core before any "the FPGA decoded a
# frame" claim is admissible.
decode_stub

[constant_inputs]

recon_u: driven by a constant-128 generate loop in stream_path.
recon_v: driven by a constant-128 generate loop in stream_path.
dpb_write_base: tied to 32'd0; no DPB frame-buffer allocator feeds the core.

# --- P16x16 zero-MV path: present in the core, unreachable from stream_path ---
p16_zero_mv_valid: tied to 1'b0; the P16 path inside the core can never fire in product.
p16_mb_x: tied to 8'd0; no P16 walker drives the core seam.

# --- DPB memory: the core read port is answered with zeros ---
dpb_rd_data: tied to 8'd0; the core has no memory behind its DPB read port.
